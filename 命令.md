# Bazel 下载缓存保留、编译输出冷构建命令

```bash
cd /data/sonic/sonic-dzf-time-measure

OUTPUT_BASE="$PWD/.measure/20260901-233744/bazel-fully-cold-attempt3/output-base"
REPOSITORY_CACHE="$PWD/.measure/20260901-233744/bazel-fully-cold-attempt3/repository-cache"
EVIDENCE="$PWD/.measure/manual-bazel-recording-$(date +%Y%m%d-%H%M%S)"
mkdir -p "$EVIDENCE"

bazel --output_base="$OUTPUT_BASE" shutdown

bazel --output_base="$OUTPUT_BASE" fetch \
  --repository_cache="$REPOSITORY_CACHE" \
  //platform/vs:sonic-vs-bin

du -sh "$OUTPUT_BASE/external" "$REPOSITORY_CACHE"

bazel --output_base="$OUTPUT_BASE" clean

rm -f target/debs/trixie/p4lang-*
rm -rf src/p4lang/p4lang-pi-0.1.1 \
       src/p4lang/p4lang-bmv2-1.15.0 \
       src/p4lang/p4lang-p4c-1.2.4.2

bazel --output_base="$OUTPUT_BASE" shutdown

/usr/bin/time -v bazel --output_base="$OUTPUT_BASE" build \
  --repository_cache="$REPOSITORY_CACHE" \
  --repository_disable_download \
  --remote_cache= \
  --experimental_remote_downloader= \
  --disk_cache= \
  --action_env=BUILD_SKIP_TEST=y \
  --action_env=HTTP_PROXY=http://127.0.0.1:9 \
  --action_env=HTTPS_PROXY=http://127.0.0.1:9 \
  --action_env=http_proxy=http://127.0.0.1:9 \
  --action_env=https_proxy=http://127.0.0.1:9 \
  --profile="$EVIDENCE/profile.gz" \
  --build_event_json_file="$EVIDENCE/bep.json" \
  --show_progress_rate_limit=30 \
  //platform/vs:sonic-vs-bin
```
